javascript/* ── HELPERS ── */
const today = new Date(); today.setHours(0,0,0,0);

function daysLeft(due) {
    if (!due || due === 'None' || !due.trim()) return null;
    const d = new Date(due + 'T00:00:00');
    if (isNaN(d)) return null;
    return Math.ceil((d - today) / 86400000);
}

function dotColorClass(rem) {
    if (rem === null) return 's-grey';
    if (rem <= 0)   return 's-red';
    if (rem <= 60)  return 's-orange';
    if (rem <= 120) return 's-yellow';
    return 's-green';
}

function badgeColorClass(rem) {
    if (rem === null) return 'ib-gy';
    if (rem <= 0)   return 'ib-r';
    if (rem <= 60)  return 'ib-o';
    if (rem <= 120) return 'ib-y';
    return 'ib-g';
}

function fmtDate(due) {
    if (!due || due === 'None') return 'ไม่มีข้อมูล';
    const d = new Date(due + 'T00:00:00');
    return isNaN(d) ? 'ไม่มีข้อมูล'
        : d.toLocaleDateString('th-TH', { year:'numeric', month:'short', day:'numeric' });
}

function buildTip(label, num, due) {
    const rem = daysLeft(due);
    const st  = rem === null ? 'ไม่มีข้อมูล'
              : rem <= 0    ? '⚠ หมดอายุแล้ว'
              : `เหลือ ${rem} วัน`;
    return `${label}  |  ${num || '—'}\nหมดอายุ : ${fmtDate(due)}\n${st}`;
}

/* ── INIT DOTS (เดิม) ── */
function initDots() {
    document.querySelectorAll('.sdot').forEach(dot => {
        const rem = daysLeft(dot.dataset.due);
        dot.classList.add(dotColorClass(rem));
        dot.setAttribute('data-tip', buildTip(
            dot.dataset.label, dot.dataset.num, dot.dataset.due));
    });
}

/* ── INIT ISO BADGES (ใหม่) ── */
function initIsoBadges() {
    document.querySelectorAll('.ibadge').forEach(badge => {
        const rem = daysLeft(badge.dataset.due);
        badge.classList.add(badgeColorClass(rem));

        // inject dot color
        const dot = badge.querySelector('.idot');
        const colors = { 'ib-g':'#51b74b','ib-y':'#e6b800','ib-o':'#fa6d09','ib-r':'#e74c3c','ib-gy':'#b0bec5' };
        const cls = badgeColorClass(rem);
        if (dot) dot.style.background = colors[cls] || '#b0bec5';

        badge.setAttribute('data-tip', buildTip(
            'ISO ' + badge.dataset.std, badge.dataset.num, badge.dataset.due));
    });
}

/* ── FILTER CHIPS ── */
const activeFilters = new Set();

function toggleChip(el) {
    const key = el.dataset.key;
    el.classList.toggle('active');
    activeFilters.has(key) ? activeFilters.delete(key) : activeFilters.add(key);
    applyFilters();
}

function clearChips() {
    activeFilters.clear();
    document.querySelectorAll('.fc-chip').forEach(c => c.classList.remove('active'));
    applyFilters();
}

function applyFilters() {
    document.querySelectorAll('#tableBody tr').forEach(row => {
        if (activeFilters.size === 0) {
            row.dataset.chipHide = '0';
            return;
        }
        const pass = [...activeFilters].every(key => {
            // ISO standards — ตรวจจาก data attribute ของ row
            if (key.startsWith('iso')) {
                return row.dataset[key] === 'y';
            }
            // dot-based: mit, tis, tisi, fs
            return row.dataset[key] === 'y';
        });
        row.dataset.chipHide = pass ? '0' : '1';
    });
    updateCount();
    currentPage = 1;
    paginate();
}

function updateCount() {
    const all  = document.querySelectorAll('#tableBody tr').length;
    const show = [...document.querySelectorAll('#tableBody tr')]
        .filter(r => r.dataset.chipHide !== '1' && r.dataset.hide !== '1').length;
    document.getElementById('fcVisible').textContent = show;
    document.getElementById('fcTotal').textContent   = all;
}

/* ── SEARCH (เดิม แต่ update count ด้วย) ── */
function globalSearch() {
    const q = document.getElementById('searchInput').value.toLowerCase().trim();
    document.querySelectorAll('#tableBody tr').forEach(r => {
        r.dataset.hide = r.innerText.toLowerCase().includes(q) ? '0' : '1';
    });
    currentPage = 1;
    paginate();
    updateCount();
}

/* ── SORT (เดิม) ── */
let sortDir = true;
function sortTable(n) {
    const tbody = document.getElementById('tableBody');
    const rows  = Array.from(tbody.querySelectorAll('tr'));
    rows.sort((a, b) => {
        const x = a.cells[n].innerText.trim().toLowerCase();
        const y = b.cells[n].innerText.trim().toLowerCase();
        const nx = parseFloat(x), ny = parseFloat(y);
        if (!isNaN(nx) && !isNaN(ny)) return sortDir ? nx - ny : ny - nx;
        return sortDir ? x.localeCompare(y, 'th') : y.localeCompare(x, 'th');
    });
    rows.forEach(r => tbody.appendChild(r));
    sortDir = !sortDir;
    paginate();
}

/* ── PAGINATION (เดิม) ── */
let currentPage = 1, rowsPerPage = 10;

function vis() {
    return [...document.querySelectorAll('#tableBody tr')]
        .filter(r => r.dataset.hide !== '1' && r.dataset.chipHide !== '1');
}

function paginate() {
    const rows = vis();
    const max  = Math.max(1, Math.ceil(rows.length / rowsPerPage));
    if (currentPage > max) currentPage = max;
    const s = (currentPage - 1) * rowsPerPage, e = s + rowsPerPage;
    document.querySelectorAll('#tableBody tr').forEach(r => r.style.display = 'none');
    rows.forEach((r, i) => r.style.display = (i >= s && i < e) ? '' : 'none');
    renderPg(max);
}

function renderPg(max) {
    const c = document.getElementById('paginationNumbers');
    c.innerHTML = '';
    const add = (lbl, en, fn, active = false) => {
        if (lbl === '…') {
            const s = document.createElement('span');
            s.className = 'pg-ell'; s.textContent = '…';
            c.appendChild(s); return;
        }
        const b = document.createElement('button');
        b.className = 'pg-btn' + (active ? ' pg-on' : '');
        b.textContent = lbl; b.disabled = !en;
        if (en) b.onclick = fn;
        c.appendChild(b);
    };
    add('‹', currentPage > 1, () => go(currentPage - 1));
    const rng = (() => {
        if (max <= 7) return Array.from({length: max}, (_, i) => i + 1);
        if (currentPage <= 4) return [1,2,3,4,5,'…',max];
        if (currentPage >= max-3) return [1,'…',max-4,max-3,max-2,max-1,max];
        return [1,'…',currentPage-1,currentPage,currentPage+1,'…',max];
    })();
    rng.forEach(v => v === '…' ? add('…') : add(v, true, () => go(v), v === currentPage));
    add('›', currentPage < max, () => go(currentPage + 1));
}

function go(n) { currentPage = n; paginate(); }

function changePageSize() {
    rowsPerPage = +document.getElementById('rowsPerPage').value;
    currentPage = 1; paginate();
}

/* ── EXPORT ── */
function exportExcel() {
    const b = new Blob(['\ufeff' + document.getElementById('productTable').outerHTML],
        { type: 'application/vnd.ms-excel;charset=utf-8' });
    Object.assign(document.createElement('a'),
        { href: URL.createObjectURL(b), download: 'products.xls' }).click();
}
function exportPDF() { window.print(); }

        r.addEventListener('click', () => location.href = r.dataset.href);
    });
}
